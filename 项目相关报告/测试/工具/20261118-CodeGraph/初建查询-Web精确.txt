4	export function isRectInVisibleViewport(rect: DOMRect, container: DOMRect): boolean {
5	  const viewport = window.visualViewport;
6	  const top = viewport?.offsetTop ?? 0, left = viewport?.offsetLeft ?? 0;
7	  const bottom = top + (viewport?.height ?? window.innerHeight), right = left + (viewport?.width ?? window.innerWidth);
8	  return rect.height > 0 && rect.width > 0 && rect.top >= Math.max(top, container.top) && rect.bottom <= Math.min(bottom, container.bottom)
9	    && rect.left >= Math.max(left, container.left) && rect.right <= Math.min(right, container.right);
10	}

12	export function useVisibleViewport() {
13	  useLayoutEffect(() => {
14	    const root = document.documentElement;
15	    const viewport = window.visualViewport;
16	    let frame = 0;
17	    const update = () => {
18	      // Pinch zoom must keep its document layout rather than reflow every gesture.
19	      if (viewport && viewport.scale !== 1) return;
20	      const height = viewport?.height ?? window.innerHeight;
21	      if (!Number.isFinite(height) || height <= 0) return;
22	      root.style.setProperty("--visible-height", `${height}px`);
23	      root.style.setProperty("--visible-top", `${Math.max(0, viewport?.offsetTop ?? 0)}px`);
24	      const editing = document.activeElement?.matches("input, textarea, [contenteditable='true']");
25	      root.dataset.compact = height < 500 ? "true" : "false";
26	      // Some mobile browsers shrink both viewports when the keyboard opens.
27	      root.dataset.keyboard = editing && (window.innerHeight - height > 100 || height < 500) ? "open" : "closed";
28	    };
29	    const schedule = () => {
30	      cancelAnimationFrame(frame);
31	      frame = requestAnimationFrame(update);
32	    };
33	    update();
34	    viewport?.addEventListener("resize", schedule);
35	    viewport?.addEventListener("scroll", schedule);
36	    window.addEventListener("resize", schedule);
37	    document.addEventListener("focusin", schedule);
38	    document.addEventListener("focusout", schedule);
39	    return () => {
40	      cancelAnimationFrame(frame);
41	      viewport?.removeEventListener("resize", schedule);
42	      viewport?.removeEventListener("scroll", schedule);
43	      window.removeEventListener("resize", schedule);
44	      document.removeEventListener("focusin", schedule);
