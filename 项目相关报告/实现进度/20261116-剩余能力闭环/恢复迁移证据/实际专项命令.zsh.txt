#!/bin/zsh
# shell: zsh
FINISHING=0
finish_regression() {
    local code="$1"
    (( ZSH_SUBSHELL == 0 && FINISHING == 0 )) || return "$code"
    FINISHING=1
    set +e
    local child
    for child in "${APP_PID:-}" "${NATS_PID:-}"; do
        if [[ -n "$child" ]] && kill -0 "$child" 2>/dev/null; then kill -TERM "$child"; wait "$child"; fi
    done
    if [[ -n "${RUN:-}" && -n "${PG_BIN:-}" ]]; then
        local label
        for label in Journal Main; do
            if [[ -f "$RUN/${label}Database/PG_VERSION" ]]; then
                "$PG_BIN/pg_ctl" -D "$RUN/${label}Database" -m fast -w -t 30 stop >> "$REPORT/夹具生命周期.log" 2>&1
            fi
        done
    fi
    printf 'fixtureExit=%s\nprivateRun=%s\n' "$code" "${RUN:-not-created}" >> "$REPORT/夹具结果.txt"
    print "fixtureExit=$code report=$REPORT"
    return "$code"
}
TRAPEXIT() { finish_regression "$?"; }
TRAPZERR() { local failed=$?; finish_regression "$failed"; exit "$failed"; }
TRAPINT() { exit 130; }
TRAPTERM() { exit 143; }
